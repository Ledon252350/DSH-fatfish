定义一条路线

const whale = document.querySelector('fatfish');

// 路线 = 若干段跳跃，每段给起跳点和落点；高度不写就按水平距离算
whale.routeFor = (host, size) => {
  const r = host.querySelector('.thinking').getBoundingClientRect();
  const h = host.getBoundingClientRect();
  const x = v => v - h.left, y = v => v - h.top;
  const water = host.clientHeight - 70;         // 水线
  const top = y(r.top) - 8;                     // 块的上方
  return [
    { from: { x: x(r.left) - 46,  y: water }, to: { x: x(r.left) + r.width / 2, y: top } },
    { from: { x: x(r.left) + r.width / 2, y: top }, to: { x: x(r.right) + 46, y: water } },
  ];
};

属性

属性作用默认

只作用于网页 DOM。桌面 App 和系统窗口盖不到，那种需求要 Electron 透明置顶窗口或系统级覆盖层。系统开了“减少动态效果”时静止，不出水花。
